Etapa 5 · Componentes

TaskApp: o dono do estado

O CRUD de tasks com useState: cria, marca e exclui chamando as APIs que já existiam.

src/resources/js/components/TaskApp.tsx
+ import { useState } from "react"
+ import type { Task } from "../types/api"
+ import { taskCreateApi } from "../api/taskCreateApi"
+ import { taskUpdateApi } from "../api/taskUpdateApi"
+ import { taskDeleteApi } from "../api/taskDeleteApi"
+ import { TaskItem } from "./TaskItem"
+
+ type TaskAppProps = {
+ idUser: number
+ initialTasks: Task[]
+ }
+
+ export function TaskApp({ idUser, initialTasks }: TaskAppProps) {
+ const [tasks, setTasks] = useState<Task[]>(initialTasks)
+ const [name, setName] = useState("")
+
+ const pending = tasks.filter(task => !task.is_done).length
+
+ // C: create
+ async function create() {
+ if (name.trim() === "") return
+
+ const created = await taskCreateApi(idUser, { name })
+ setTasks(current => [...current, created])
+ setName("")
+ }
+
+ // U: update (marcar/desmarcar como feita)
+ async function toggle(task: Task) {
+ const updated = await taskUpdateApi(idUser, task.id, { is_done: !task.is_done })
+ setTasks(current => current.map(item => {
+ if (item.id !== updated.id) return item
+ return updated
+ }))
+ }
+
+ // D: delete
+ async function remove(id: number) {
+ await taskDeleteApi(idUser, id)
+ setTasks(current => current.filter(task => task.id !== id))
+ }
+
+ return (
+ <section>
+ <div className="input-group mb-3">
+ <input
+ className="form-control"
+ value={name}
+ onChange={event => setName(event.target.value)}
+ />
+ <button className="btn btn-primary" onClick={create}>Adicionar</button>
+ </div>
+
+ <p>{pending} pendente(s)</p>
+
+ <ul className="list-group">
+ {tasks.map(task => (
+ <TaskItem key={task.id} task={task} onToggle={toggle} onDelete={remove} />
+ ))}
+ </ul>
+ </section>
+ )
+ }

Array novo

[...current, created], map e filter criam um endereço novo: é isso que o React compara.

Forma de função

As APIs são assíncronas; setTasks(current => ...) parte do valor mais recente.

O modal: estados e a busca

O AssignTaskModal guarda tudo em estado e a busca é um handler explícito (handleSearch), sem useEffect.

src/resources/js/components/AssignTaskModal.tsx (estados e busca)
+ import { useState } from "react"
+ import type { FormEvent } from "react"
+ import type { Task, User, UserSearchFilters } from "../types/api"
+ import { userSearchApi } from "../api/userSearchApi"
+ import { taskCreateApi } from "../api/taskCreateApi"
+ import { getErrorMessage } from "../utils/getErrorMessage"
+ import { UserSearchResults } from "./UserSearchResults"
+
+ type SearchField = "name" | "email" | "job_title" | "city" | "state" | "bio"
+
+ const FIELDS: { value: SearchField, label: string }[] = [
+ { value: "name", label: "Nome" },
+ { value: "email", label: "E-mail" },
+ { value: "job_title", label: "Cargo" },
+ { value: "city", label: "Cidade" },
+ { value: "state", label: "UF" },
+ { value: "bio", label: "Bio" },
+ ]
+
+ type AssignTaskModalProps = {
+ isOpen: boolean
+ onClose: () => void
+ onCreated: (task: Task, user: User) => void
+ }
+
+ export function AssignTaskModal({ isOpen, onClose, onCreated }: AssignTaskModalProps) {
+ const [taskName, setTaskName] = useState("")
+ const [field, setField] = useState<SearchField>("name")
+ const [text, setText] = useState("")
+ const [users, setUsers] = useState<User[] | null>(null)
+ const [total, setTotal] = useState(0)
+ const [selected, setSelected] = useState<User | null>(null)
+ const [error, setError] = useState("")
+ const [isSearching, setIsSearching] = useState(false)
+ const [isCreating, setIsCreating] = useState(false)
+
+ const canCreate = taskName.trim() !== "" && selected !== null && !isCreating
+
+ function reset() {
+ setTaskName("")
+ setField("name")
+ setText("")
+ setUsers(null)
+ setTotal(0)
+ setSelected(null)
+ setError("")
+ }
+
+ function handleClose() {
+ reset()
+ onClose()
+ }
+
+ // Buscar: botão ou Enter (o campo está num <form>)
+ async function handleSearch(event: FormEvent) {
+ event.preventDefault()
+
+ if (text.trim() === "") {
+ setError("Digite algo para buscar")
+ return
+ }
+
+ const filters: UserSearchFilters = {}
+ filters[field] = text.trim()
+
+ setError("")
+ setIsSearching(true)
+
+ try {
+ const result = await userSearchApi(filters, 1, 10)
+ setUsers(result.data)
+ setTotal(result.total)
+ } catch (error) {
+ setError(getErrorMessage(error, "Erro ao buscar usuários"))
+ }
+
+ setIsSearching(false)
+ }
+

Estados com nome

isSearching e isCreating são booleanos; users começa null para saber que ainda não buscou.

Texto vazio

Não chama a API: mostra "Digite algo para buscar". O Enter funciona porque o campo está num <form>.

★ Toda página é uma View protegida pelo JWT

A regra da aula: nenhuma página estática. Cada tela é uma rota de View que passa pelo middleware ANTES de o HTML existir.

GET /tasks · GET /usuarios
→ RedirectIfGuestMiddleware // jwt.verify: assinatura + validade
sem cookie, token inválido ou expirado → redirect /login
token ok → request.user = { id }
→ controller (usa request.user, nunca dado do cliente)
→ res.render("tasks" | "users", ...)
→ HTML com #root → navegador → createRoot(...) → componente React

→ RedirectIfAuthenticatedMiddleware // também jwt.verify
token ok → redirect /tasks · senão renderiza o login
